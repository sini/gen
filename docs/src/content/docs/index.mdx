## The wiring problem

Take a homelab with two machines. Both run SSH. The server also exports metrics, and the laptop runs
a desktop. In plain NixOS the wiring looks like this:

```nix
nixosConfigurations = {
  laptop = nixpkgs.lib.nixosSystem {
    specialArgs = { addr = "10.0.0.2"; users = [ "alice" ]; };
    modules = [ ./ssh.nix ./desktop.nix ];
  };
  server = nixpkgs.lib.nixosSystem {
    specialArgs = { addr = "10.0.0.1"; users = [ "alice" "bob" ]; };
    modules = [ ./ssh.nix ./monitoring.nix ]; # monitoring.nix needs ssh.nix
  };
};
```

Three facts in that file live only in your head:

- `monitoring.nix` needs `ssh.nix`. A host that lists the first and forgets the second still
  evaluates.
- `addr` and `users` reach every module through `specialArgs`, whether the module reads them or not.
- The module lists are the only record of which host has which feature. To answer "which hosts run
  SSH?" you read every list.

## The same homelab in gen

Each feature is an *aspect*, a named bundle of configuration. An aspect can include other aspects,
and each include is an edge in a graph. An aspect that needs to know its host takes `host` as an
argument, and gen passes the host in.

```nix
aspects.ssh.nixos = { host, ... }: {
  services.openssh.enable = true;
  services.openssh.settings.AllowUsers = host.users;
};
aspects.desktop.nixos.services.xserver.enable = true;
aspects.monitoring.includes = [ "ssh" ];
aspects.monitoring.nixos = { host, ... }: {
  services.prometheus.exporters.node.listenAddress = host.addr;
};

hosts.laptop = { addr = "10.0.0.2"; users = [ "alice" ]; aspects = [ "desktop" "ssh" ]; };
hosts.server = { addr = "10.0.0.1"; users = [ "alice" "bob" ]; aspects = [ "monitoring" ]; };
```

The server lists only `monitoring`, and SSH arrives through the include edge. Each host's NixOS
configuration comes out filled in with its own address and users (output formatted for reading):

```console
$ nix eval --file homelab.nix
{
  includesOf = { desktop = [ ]; monitoring = [ "ssh" ]; ssh = [ ]; };
  laptop.services = {
    openssh = { enable = true; settings.AllowUsers = [ "alice" ]; };
    xserver.enable = true;
  };
  server.services = {
    openssh = { enable = true; settings.AllowUsers = [ "alice" "bob" ]; };
    prometheus.exporters.node.listenAddress = "10.0.0.1";
  };
}
```

<details>
<summary>The whole of `homelab.nix`</summary>

This file runs as it is, with no flake and no nixpkgs. The last step evaluates each host's modules
with gen's own module system. In a real flake that step is a NixOS system builder.

```nix
let
  gen = import (builtins.fetchGit "https://github.com/sini/gen") { };
  m = gen.merge;
  t = m.types;

  cnf.keySemantics.nixos.category = "class";
  aspectSchema = gen.aspects.mkAspectSchema cnf;

  declared = m.evalModuleTree {
    modules = [
      (aspectSchema.mkAspectModule { })
      { options.hosts = m.mkOption { type = t.attrsOf t.raw; }; }
      {
        aspects.ssh.nixos =
          { host, ... }:
          {
            services.openssh.enable = true;
            services.openssh.settings.AllowUsers = host.users;
          };
        aspects.desktop.nixos.services.xserver.enable = true;
        aspects.monitoring.includes = [ "ssh" ];
        aspects.monitoring.nixos =
          { host, ... }:
          {
            services.prometheus.exporters.node.listenAddress = host.addr;
          };

        hosts.laptop = {
          addr = "10.0.0.2";
          users = [ "alice" ];
          aspects = [ "desktop" "ssh" ];
        };
        hosts.server = {
          addr = "10.0.0.1";
          users = [ "alice" "bob" ];
          aspects = [ "monitoring" ];
        };
      }
    ];
  };

  # The include edges, and each host's aspects plus everything they reach.
  facts = gen.aspects.graphFacts cnf declared.config.aspects;
  withIncludes =
    names: names ++ builtins.concatMap (gen.graph.reachableFrom { edges = a: facts.includesOf.${a}; }) names;

  projected = gen.delivery.project {
    values = declared.config;
    inherit cnf;
    selectNodes = v: builtins.mapAttrs (_: h: h // { aspects = withIncludes h.aspects; }) v.hosts;
  };

  # Each host's `nixos` modules, with `host` bound to that host's record.
  realized = gen.delivery.realize {
    inherit projected;
    terminals.nixos =
      { modules, bindings, ... }:
      (m.evalModuleTree {
        modules = [
          { freeformType = t.lazyAttrsOf t.anything; }
        ] ++ map (mod: (gen.bind.wrap { bindings.host = bindings.node; } mod).module) modules;
      }).config;
  };
in
{
  inherit (facts) includesOf;
  inherit (realized.nixos) laptop server;
}
```
